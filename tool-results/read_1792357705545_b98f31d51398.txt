    81→    81→/**
    82→    82→ * Greatest bar open-time (UNIX seconds) at-or-before targetMs.
    83→    83→ * Snapping keeps markers placeable even when the analysis interval differs
    84→    84→ * from the chart interval (larger intervals align to smaller boundaries).
    85→    85→ */
    86→    86→function snapBarTimeSec(candles: Candle[], targetMs: number, intervalMs: number): number | null {
    87→    87→  if (candles.length === 0) return null
    88→    88→  const first = candles[0]
    89→    89→  const last = candles[candles.length - 1]
    90→    90→  if (targetMs < first.openTime - intervalMs) return null
    91→    91→  if (targetMs > last.openTime + intervalMs) return null
    92→    92→  let lo = 0
    93→    93→  let hi = candles.length - 1
    94→    94→  let ans = 0
    95→    95→  while (lo <= hi) {
    96→    96→    const mid = (lo + hi) >> 1
    97→    97→    if (candles[mid].openTime <= targetMs) {
    98→    98→      ans = mid
    99→    99→      lo = mid + 1
   100→   100→    } else {
   101→   101→      hi = mid - 1
   102→   102→    }
   103→   103→  }
   104→   104→  return Math.floor(candles[ans].openTime / 1000)
   105→   105→}

   155→   155→  /* Overlay rendering (logical → pixel on every chart redraw)        */

   157→   157→
   158→   158→  const redrawOverlay = useCallback(() => {
   159→   159→    const canvas = overlayRef.current
   160→   160→    const chart = chartRef.current
   161→   161→    const series = candleRef.current
   162→   162→    if (!canvas || !chart || !series) return
   163→   163→    const ctx = canvas.getContext('2d')
   164→   164→    if (!ctx) return
   165→   165→    const w = canvas.clientWidth
   166→   166→    const h = canvas.clientHeight
   167→   167→    if (w < 2 || h < 2) return
   168→   168→
   169→   169→    const dpr = window.devicePixelRatio || 1
   170→   170→    const pw = Math.round(w * dpr)
   171→   171→    const ph = Math.round(h * dpr)
   172→   172→    if (canvas.width !== pw || canvas.height !== ph) {
   173→   173→      canvas.width = pw
   174→   174→      canvas.height = ph
   175→   175→    }
   176→   176→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   177→   177→    ctx.clearRect(0, 0, w, h)
   178→   178→
   179→   179→    const ts = chart.timeScale()
   180→   180→    const input: OverlayRenderInput = {
   181→   181→      w,
   182→   182→      h,
   183→   183→      xOfTime: (t) => ts.timeToCoordinate(t as UTCTimestamp),
   184→   184→      yOfPrice: (p) => series.priceToCoordinate(p),
   185→   185→      palette: readPalette(),
   186→   186→      intervalMs: intervalMsRef.current,
   187→   187→    }

   189→   189→    const draft = draftRef.current
   190→   190→    if (draft) renderDrawing(ctx, { ...input, isPreview: true }, draft)
   191→   191→  }, [])
   192→   192→
   193→   193→  /**
   194→   194→   * Signature of everything that affects overlay geometry (visible logical
   195→   195→   * range, price mapping of the last close, canvas size, drawings revision,
   196→   196→   * draft position). A rAF watchdog compares it every frame — this catches
   197→   197→   * price-axis drags & auto-rescales which have no dedicated event.
   198→   198→   */
   199→   199→  const overlaySignature = useCallback(() => {
   200→   200→    const chart = chartRef.current
   201→   201→    const series = candleRef.current
   202→   202→    const canvas = overlayRef.current
   203→   203→    if (!chart || !series || !canvas) return 'no'
   204→   204→    const lr = chart.timeScale().getVisibleLogicalRange()
   205→   205→    const last = candlesRef.current.length ? candlesRef.current[candlesRef.current.length - 1] : null
   206→   206→    const y = last ? series.priceToCoordinate(last.close) : null
   207→   207→    const x = last ? chart.timeScale().timeToCoordinate((last.openTime / 1000) as UTCTimestamp) : null
   208→   208→    const draft = draftRef.current
   209→   209→    return [
   210→   210→      canvas.clientWidth,
   211→   211→      canvas.clientHeight,
   212→   212→      lr ? `${lr.from.toFixed(2)}:${lr.to.toFixed(2)}` : '-',
   213→   213→      y != null ? y.toFixed(1) : '-',
   214→   214→      x != null ? x.toFixed(1) : '-',
   215→   215→      drawingsRevRef.current,
   216→   216→      draft ? `${draft.time2}:${draft.price2}` : '-',
   217→   217→    ].join('|')
   218→   218→  }, [])
   219→   219→
   220→   220→  /* ---------------------------------------------------------------- */
   221→   221→  /* Chart lifecycle (create once, theme/resize observers, cleanup)   */

   286→   286→    mo.observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] })
   287→   287→
   288→   288→    // rAF watchdog — redraw the overlay whenever the visual mapping changed
   289→   289→    let raf = 0
   290→   290→    let lastSig = ''
   291→   291→    const tick = () => {
   292→   292→      const sig = overlaySignature()

   294→   294→        lastSig = sig
   295→   295→        redrawOverlay()
   296→   296→      }
   297→   297→      raf = requestAnimationFrame(tick)
   298→   298→    }
   299→   299→    raf = requestAnimationFrame(tick)
   300→   300→
   301→   301→    return () => {
   302→   302→      cancelAnimationFrame(raf)
   303→   303→      ro.disconnect()
   304→   304→      mo.disconnect()
   305→   305→      markersApi.detach()
   306→   306→      chart.remove()
   307→   307→      chartRef.current = null
   308→   308→      candleRef.current = null
   309→   309→      volumeRef.current = null
   310→   310→      markersApiRef.current = null
   311→   311→    }
   312→   312→  }, [redrawOverlay, overlaySignature])
   313→   313→
   314→   314→  /* ---------------------------------------------------------------- */
   315→   315→  /* Markers & price lines                                             */
   316→   316→  /* ---------------------------------------------------------------- */
   317→   317→
   318→   318→  const refreshMarkers = useCallback(() => {
   319→   319→    const api = markersApiRef.current
   320→   320→    if (!api) return
   321→   321→    const an = analysisRef.current
   322→   322→    const candles = candlesRef.current
   323→   323→    const list: SeriesMarker<Time>[] = []
   324→   324→    if (an && signalsVisibleRef.current && candles.length > 0) {
   325→   325→      const intervalMs = intervalMsRef.current
   326→   326→      const pendingId = an.pendingSignal?.id
   327→   327→      const seen = new Set<string>()
   328→   328→      for (const s of an.signals ?? []) {
   329→   329→        if (seen.has(s.id)) continue
   330→   330→        seen.add(s.id)
   331→   331→        const t = snapBarTimeSec(candles, s.candleOpenTime, intervalMs)
   332→   332→        if (t == null) continue
   333→   333→        const isPending = pendingId != null && s.id === pendingId
   334→   334→        const isLong = s.side === 'Long'
   335→   335→        list.push({
   336→   336→          time: t as UTCTimestamp,
   337→   337→          position: isLong ? 'belowBar' : 'aboveBar',
   338→   338→          color: isLong ? UP_COLOR : DOWN_COLOR,
   339→   339→          shape: isLong ? 'arrowUp' : 'arrowDown',
   340→   340→          text: isPending ? (isLong ? 'L · در انتظار' : 'S · در انتظار') : isLong ? 'L' : 'S',
   341→   341→          size: isPending ? 2 : 1,
   342→   342→        })
   343→   343→      }
   344→   344→      list.sort((a, b) => (a.time as number) - (b.time as number))
   345→   345→    }
   346→   346→    api.setMarkers(list)
   347→   347→  }, [])
   348→   348→
   349→   349→  const refreshPriceLines = useCallback(() => {

   355→   355→      } catch {
   356→   356→        /* already detached */
   357→   357→      }
   358→   358→    }
   359→   359→    priceLinesRef.current = []
   360→   360→    const pending = analysisRef.current?.pendingSignal
   361→   361→    if (!pending || !signalsVisibleRef.current) return
   362→   362→    const mk = (price: number, color: string, style: LineStyle, title: string): IPriceLine =>
   363→   363→      series.createPriceLine({
   364→   364→        price,
   365→   365→        color,
   366→   366→        lineStyle: style,
   367→   367→        lineWidth: 1,
   368→   368→        title,
   369→   369→        axisLabelVisible: true,
   370→   370→      })
   371→   371→    priceLinesRef.current = [
   372→   372→      mk(pending.price, ENTRY_COLOR, LineStyle.Solid, 'Entry'),
   373→   373→      mk(pending.tp, UP_COLOR, LineStyle.Dashed, 'TP'),
   374→   374→      mk(pending.sl, DOWN_COLOR, LineStyle.Dashed, 'SL'),
   375→   375→    ]
   376→   376→  }, [])

   384→   384→      candlesRef.current = candles

   386→   386→      const series = candleRef.current

   389→   389→        series.setData(
   390→   390→          candles.map((c) => ({

   392→   392→            open: c.open,
   393→   393→            high: c.high,

   399→   399→          candles.map((c) => ({
   400→   400→            time: Math.floor(c.openTime / 1000) as UTCTimestamp,

   402→   402→            color: c.close >= c.open ? UP_VOLUME : DOWN_VOLUME,

   408→   408→      }
   409→   409→      setHoverCandle(candles.length ? candles[candles.length - 1] : null)
   410→   410→      setLastClose(candles.length ? candles[candles.length - 1].close : null)
   411→   411→      refreshMarkers()
   412→   412→      refreshPriceLines()
   413→   413→      redrawOverlay()
   414→   414→    },

   422→   422→      if (an) {
   423→   423→        for (const c of an.candles ?? []) {

   428→   428→      setSignalState(
   429→   429→        an
   430→   430→          ? { signals: an.signals ?? [], pending: an.pendingSignal ?? null, analysisInterval: an.interval ?? null }
   431→   431→          : null,
   432→   432→      )
   433→   433→      refreshMarkers()
   434→   434→      refreshPriceLines()
   435→   435→      redrawOverlay()
   436→   436→    },
   437→   437→    [refreshMarkers, refreshPriceLines, redrawOverlay],
   438→   438→  )
   439→   439→
   440→   440→  /* ---------------------------------------------------------------- */
   441→   441→  /* Data loading (klines → analysis → markers/price-lines → redraw)  */
   442→   442→  /* ---------------------------------------------------------------- */
   443→   443→
   444→   444→  useEffect(() => {
   445→   445→    const myId = ++loadIdRef.current
   446→   446→    let cancelled = false
   447→   447→    const run = async () => {
   448→   448→      setLoading(true)
   449→   449→      setError(null)
   450→   450→      try {
   451→   451→        const kl = await apiGet<KlinesResponse>(
   452→   452→          `/api/market/klines?symbol=${encodeURIComponent(symbol)}&interval=${interval}&limit=500`,
   453→   453→        )
   454→   454→        if (cancelled || myId !== loadIdRef.current) return
   455→   455→        const candles = (kl.candles ?? []).slice().sort((a, b) => a.openTime - b.openTime)
   456→   456→        if (candles.length === 0) throw new Error('کندلی برای این نماد یافت نشد')
   457→   457→        applyKlines(candles)
   458→   458→        // analysis is secondary — its failure must not blank the candles
   459→   459→        try {
   460→   460→          const an = await apiGet<AnalysisResponse>(`/api/market/analysis?symbol=${encodeURIComponent(symbol)}`)
   461→   461→          if (cancelled || myId !== loadIdRef.current) return
   462→   462→          applyAnalysis(an)
   463→   463→        } catch {
