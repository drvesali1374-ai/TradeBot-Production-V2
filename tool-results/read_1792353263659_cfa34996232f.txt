     1→/**
     2→ * Server-Side Cycle Engine
     3→ * =========================
     4→ *
     5→ * A faithful TypeScript port of the client-side `AutomationManager.runCycle()`
     6→ * logic (from public/js/automation-manager.js) so that a full automation cycle
     7→ * can run on the server — triggered either manually via /api/run-cycle or
     8→ * automatically via the Cloudflare Cron Worker (every 5 minutes).
     9→ *
    10→ * CRITICAL DESIGN RULES:
    11→ *  - The logic MUST exactly mirror the client-side runCycle() so that
    12→ *    behaviour is identical whether triggered from the browser button
    13→ *    or from the cron worker. Do not "improve" or refactor the logic.
    14→ *  - All settings are read from the permanent database (AutomationState),
    15→ *    NOT from localStorage. This is the key Cloudflare migration requirement.
    16→ *  - Indicator math (RSI/ATR/SMA, signal generation) is copied verbatim
    17→ *    from public/js/shared/signal-utils.js and analyzeMarketData().
    18→ *  - Risk controls (safe asset, price distance, max margin, positive margin)
    19→ *    use the same formulas and produce the same Bale notifications.
    20→ *
    21→ * Flow (mirrors runCycle() 14 steps):
    22→ *   1. selectNextSymbol()   — pick first ready symbol (respects tradeWaitTime)
    23→ *   2. setupForCycle()      — clear per-cycle state
    24→ *   3. fetchMarketData()    — GET /api/toobit-proxy (klines)
    25→ *   4. analyzeMarketData()  — generate signals (RSI/ATR/SMA/daily levels)
    26→ *   5. fetchPositionHistory — POST /api/history (mapped to consistent shape)
    27→ *   6. updateSelectedSignal — only latest signal with status "در انتظار"
    28→ *   7. fetchPrice()         — current price
    29→ *   8. closeOppositePositions
    30→ *   9. getSymbolMargin()    — existing same-direction margin
    31→ *  10. fetchBalance()       — total / free / positionMargin
    32→ *  11. calculateNewMargin()
    33→ *  12. Risk controls (safe asset, price distance, max margin, positive margin)
    34→ *  13. openPosition()       — POST /api/create-position
    35→ *  14. notifyOpenPosition() — POST /api/bale-send
    36→ *
    37→ * After success or skip, the symbol's status/lastCycleTime/errorCount are
    38→ * persisted back to the database.
    39→ */

   229→function calculateRSI(data: Candle[], period = 14): (number | null)[] {
   230→  let avgGain = 0
   231→  let avgLoss = 0
   232→  const rsiValues: (number | null)[] = []
   233→
   234→  if (data.length < period) {
   235→    return new Array(data.length).fill(null)
   236→  }
   237→
   238→  for (let i = 1; i < data.length; i++) {
   239→    const change = data[i].close - data[i - 1].close
   240→    const gain = Math.max(change, 0)
   241→    const loss = Math.max(-change, 0)
   242→
   243→    if (i <= period) {
   244→      avgGain += gain
   245→      avgLoss += loss
   246→      if (i === period) {
   247→        avgGain /= period
   248→        avgLoss /= period
   249→      }
   250→      rsiValues.push(null)
   251→    } else {
   252→      avgGain = (avgGain * (period - 1) + gain) / period
   253→      avgLoss = (avgLoss * (period - 1) + loss) / period
   254→      const rs = avgLoss === 0 ? 100 : avgGain / avgLoss
   255→      rsiValues.push(100 - 100 / (1 + rs))
   256→    }
   257→  }
   258→
   259→  return rsiValues
   260→}

   262→function calculateATR(data: Candle[], period = 14): number[] {
   263→  const atrValues: number[] = []
   264→
   265→  const trueRanges = data.map((candle, i) => {
   266→    if (i === 0) return candle.high - candle.low
   267→    const prevClose = data[i - 1].close
   268→    return Math.max(
   269→      candle.high - candle.low,
   270→      Math.abs(candle.high - prevClose),
   271→      Math.abs(candle.low - prevClose)
   272→    )
   273→  })
   274→
   275→  atrValues[0] = trueRanges[0]
   276→  for (let i = 1; i < data.length; i++) {
   277→    atrValues[i] = ((period - 1) * atrValues[i - 1] + trueRanges[i]) / period
   278→  }
   279→
   280→  return atrValues
   281→}
   282→
   283→function calculateSMA(data: number[], period: number): (number | null)[] {
   284→  const smaValues: (number | null)[] = []
   285→  for (let i = 0; i < data.length; i++) {
   286→    if (i < period - 1) {
   287→      smaValues.push(null)
   288→    } else {
   289→      let sum = 0
   290→      for (let j = i - period + 1; j <= i; j++) {
   291→        sum += data[j]
   292→      }
   293→      smaValues.push(sum / period)
   294→    }
   295→  }
   296→  return smaValues
   297→}

   299→function generateOrderId(timestamp: number | Date, symbol: string): string {
   300→  const dt = new Date(timestamp)
   301→  const year = dt.getFullYear().toString().padStart(4, '0')
   302→  const month = (dt.getMonth() + 1).toString().padStart(2, '0')
   303→  const day = dt.getDate().toString().padStart(2, '0')
   304→  const hour = dt.getHours().toString().padStart(2, '0')
   305→  const minute = dt.getMinutes().toString().padStart(2, '0')
   306→  const second = dt.getSeconds().toString().padStart(2, '0')
   307→  return `${symbol}_${year}${month}${day}_${hour}${minute}${second}`
   308→}

   311→// Jalali date conversion — verbatim port of toJalali() in automation-manager.js

   314→function toJalali(date?: Date | number): string {
   315→  if (!date) date = new Date()
   316→  if (typeof date === 'number') date = new Date(date)
   317→
   318→  const tehranOffset = 3.5 * 60 // minutes
   319→  const utcTime = date.getTime() + date.getTimezoneOffset() * 60000
   320→  const tehranTime = new Date(utcTime + tehranOffset * 60000)
   321→
   322→  const gy = tehranTime.getFullYear()
   323→  const gm = tehranTime.getMonth() + 1
   324→  const gd = tehranTime.getDate()
   325→
   326→  const g_d_m = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334]
   327→  let jy: number, jm: number, jd: number, gy2: number, days: number
   328→
   329→  gy2 = gm > 2 ? gy + 1 : gy
   330→  days =
   331→    355666 +
   332→    365 * gy +
   333→    Math.floor((gy2 + 3) / 4) -
   334→    Math.floor((gy2 + 99) / 100) +
   335→    Math.floor((gy2 + 399) / 400) +
   336→    gd +
   337→    g_d_m[gm - 1]
   338→  jy = -1595 + 33 * Math.floor(days / 12053)
   339→  days %= 12053
   340→  jy += 4 * Math.floor(days / 1461)
   341→  days %= 1461
   342→  if (days > 365) {
   343→    jy += Math.floor((days - 1) / 365)
   344→    days = (days - 1) % 365
   345→  }
   346→  if (days < 186) {
   347→    jm = 1 + Math.floor(days / 31)
   348→    jd = 1 + (days % 31)
   349→  } else {
   350→    jm = 7 + Math.floor((days - 186) / 30)
   351→    jd = 1 + ((days - 186) % 30)
   352→  }
   353→
   354→  const hours = String(tehranTime.getHours()).padStart(2, '0')
   355→  const minutes = String(tehranTime.getMinutes()).padStart(2, '0')
   356→  const seconds = String(tehranTime.getSeconds()).padStart(2, '0')
   357→
   358→  return `${jy}/${String(jm).padStart(2, '0')}/${String(jd).padStart(2, '0')} - ${hours}:${minutes}:${seconds}`
   359→}

   365→/**
   366→ * Execute one full automation cycle on the server.
   367→ *
   368→ * @param baseUrl - the base URL of the running app (for internal API calls).
   369→ *                  In local dev this is http://localhost:3000.
   370→ *                  In Cloudflare this is the deployed Pages URL.
   371→ * @param options - optional overrides (used by tests / manual triggers)
   372→ */
   373→export async function runCycle(
   374→  baseUrl: string,
   375→  options: {
   376→    symbolOverride?: string | null
   377→    triggerSource?: 'manual' | 'cron'
   378→  } = {}
   379→): Promise<CycleResult> {
   380→  const startTime = Date.now()
   381→  const logs: CycleLogEntry[] = []
   382→  // ✓ Batch logging: collect all logs in memory during the cycle, then flush
   383→  //   them to KV in a SINGLE write at the end. This avoids the race condition
   384→  //   where concurrent cron cycles overwrite each other's logs (each log was
   385→  //   previously doing a full read-modify-write of the entire logs array).
   386→  const pendingDbLogs: Array<{
   387→    message: string
   388→    type: string
   389→    symbol: string
   390→    source: string
   391→    timestamp: string
   392→  }> = []
   393→
   394→  const log = (message: string, type: CycleLogEntry['type'] = 'info', symbol?: string) => {
   395→    const entry: CycleLogEntry = { message, type, symbol, timestamp: new Date().toISOString() }
   396→    logs.push(entry)
   397→    // Buffer for batch write — do NOT write to KV on each log call
   398→    pendingDbLogs.push({
   399→      message,
   400→      type,
   401→      symbol: symbol || '',
   402→      source: options.triggerSource || 'server',
   403→      timestamp: entry.timestamp
   404→    })
   405→    console.log(`[CycleEngine${options.triggerSource ? `:${options.triggerSource}` : ''}] ${type.toUpperCase()} ${symbol ? `[${symbol}] ` : ''}${message}`)
   406→  }
   407→
   408→  // ✓ Flush all buffered logs to KV.
   409→  //   Uses ONE write strategy (Task 11 — KV write reduction):
   410→  //     1. `lastCycleLogs` — stores THIS cycle's complete logs (single KV put).
   411→  //   The shared historical `logs` array is NO LONGER updated per-cycle.
   412→  //   Instead, /api/run-cycle does ONE batch append of all cycle logs after
   413→  //   all cycles in a single cron call complete (via AutomationLogs.appendBatch).
   414→  // ✓ Transaction Buffer (پیشنهاد ۱): flushLogs دیگر به KV نمی‌نویسد
   415→  //   logs در CycleResult.logs برگردانده می‌شوند و /api/run-cycle در پایان
   416→  //   یک‌بار آنها را به lastCycleLogs و shared logs می‌نویسد
   417→  const flushLogs = async () => {
   418→    // No-op: logs are already in the `logs` array and will be returned in CycleResult
   419→    // The caller (/api/run-cycle) is responsible for writing them to KV
   420→  }
   421→
   422→  try {
   423→    // -------------------------------------------------------------------
   424→    // Load settings + symbols from the permanent database (async — KV)
   425→    // -------------------------------------------------------------------
   426→    const dbAutomationSettings = await AutomationState.get('automation_settings')
   427→    const automationSettings: AutomationSettings = {
   428→      ...getDefaultAutomationSettings(),
   429→      ...(typeof dbAutomationSettings === 'string'
   430→        ? JSON.parse(dbAutomationSettings)
   431→        : (dbAutomationSettings as Record<string, unknown>) || {})
   432→    }
   433→
   434→    const dbMarketSettings = await AutomationState.get('marketSignalSettings')
   435→    const marketSettings: MarketSignalSettings =
   436→      typeof dbMarketSettings === 'string'
   437→        ? JSON.parse(dbMarketSettings)
   438→        : (dbMarketSettings as MarketSignalSettings) || {}
   439→
   440→    const dbSymbols = await AutomationState.get('automation_symbols')
   441→    const symbols: SymbolEntry[] = (() => {
   442→      const parsed =
   443→        typeof dbSymbols === 'string' ? JSON.parse(dbSymbols) : (dbSymbols as SymbolEntry[]) || []
   444→      return Array.isArray(parsed) ? parsed : []
   445→    })()
   446→
   447→    if (symbols.length === 0) {
   448→      log('هیچ نمادی در لیست اتوماسیون وجود ندارد', 'warning')
   449→      return {
   450→        success: false,
   451→        symbol: null,
   452→        action: 'no_ready_symbol',
   453→        logs,
   454→        errorCount: 0,
   455→        duration: Date.now() - startTime,
   456→        symbols
   457→      }
   458→    }
   459→
   460→    // -------------------------------------------------------------------
   461→    // Step 1: selectNextSymbol()
   462→    // -------------------------------------------------------------------
   463→    const now = Date.now()
   464→    const waitTimeMs = automationSettings.tradeWaitTime * 60 * 1000
   465→    const readySymbols = symbols.filter((s) => {
   466→      if (!s.lastCycleTime) return true
   467→      return now - s.lastCycleTime >= waitTimeMs
   468→    })
   469→
   470→    let symbol: SymbolEntry | null = null
   471→    if (options.symbolOverride) {
   472→      symbol = symbols.find((s) => s.name === options.symbolOverride) || null
   473→    } else {
   474→      symbol = readySymbols.length > 0 ? readySymbols[0] : null
   475→    }
   476→
   477→    if (!symbol) {
   478→      log('هیچ نمادی برای معامله آماده نیست', 'warning')
   479→      return {
   480→        success: false,
   481→        symbol: null,
   482→        action: 'no_ready_symbol',
   483→        logs,
   484→        errorCount: 0,
   485→        duration: Date.now() - startTime,
   486→        symbols
   487→      }
   488→    }
   489→
   490→    log(`شروع چرخه برای نماد ${symbol.name}`, 'info', symbol.name)
   491→
   492→    // -------------------------------------------------------------------
   493→    // Helper to persist symbol state back to DB
   494→    // ✓ IMPORTANT: now async (await) — caller MUST await to ensure KV consistency
   495→    //
   496→    // ✓ Task 11 (KV write reduction): We have TWO helpers now:
   497→    //   - `updateSymbolState` (async) — does the actual KV write. Used ONLY at
   498→    //     the START of a cycle to set `status: 'running'` + `lastCycleTime`
   499→    //     (one write per cycle, persisted immediately to prevent the same
   500→    //     symbol being picked again by the next cycle).
   501→    //   - `updateSymbolStateInMemory` (sync, no KV write) — used at the END of
   502→    //     every cycle (success / no_signal / risk_blocked) to flip `status`
   503→    //     back to 'waiting'. The `lastCycleTime` was already persisted at the
   504→    //     START, so the next cycle's selectNextSymbol will skip this symbol.
   505→    //     The status change is cosmetic (UI reads it) — not worth a KV write.
   506→    //
   507→    //     The ONE exception is the catch block (error path): it does its own
   508→    //     read-modify-write of `automation_symbols` to bump `errorCount`,
   509→    //     which MUST be persisted (allowedErrors tracking). That write is
   510→    //     kept as-is — only the redundant end-of-cycle 'waiting' writes are
   511→    //     removed.
   512→    // -------------------------------------------------------------------
   513→    // ✓ Transaction Buffer (پیشنهاد ۱+۶): تمام state updates در حافظه انجام می‌شود
   514→    //   KV write فقط در پایان /api/run-cycle انجام می‌شود (نه در هر چرخه)
   515→    //   این KV writes را از ~۵ به ~۱-۲ در هر چرخه کاهش می‌دهد
   516→    const updateSymbolState = async (updates: Partial<SymbolEntry>) => {
   517→      // ✓ In-memory only — NO KV write during cycle
   518→      const idx = symbols.findIndex((s) => s.id === symbol!.id)
   519→      if (idx >= 0) {
   520→        symbols[idx] = { ...symbols[idx], ...updates }
   521→      }
   522→    }
   523→
   524→    const updateSymbolStateInMemory = (updates: Partial<SymbolEntry>) => {
   525→      const idx = symbols.findIndex((s) => s.id === symbol!.id)
   526→      if (idx >= 0) {
   527→        symbols[idx] = { ...symbols[idx], ...updates }
   528→      }
   529→    }
   530→
   531→    // ✓ FIX: Set lastCycleTime IMMEDIATELY and persist to KV right away
   532→    //   This is CRITICAL: if we only update in-memory, the NEXT cron call (5 min later)
   533→    //   will re-read the OLD lastCycleTime from KV and select the same symbol again.
   534→    //   The 1 extra KV write per cycle is worth it to ensure proper symbol rotation.
   535→    try {
   536→      const idx = symbols.findIndex((s) => s.id === symbol!.id)
   537→      if (idx >= 0) {
   538→        symbols[idx] = { ...symbols[idx], lastCycleTime: Date.now() }
   539→        await AutomationState.set('automation_symbols', symbols)
   540→      }
   541→    } catch (e) {
   542→      // KV write failed — update in-memory only (fallback)
   543→      const idx = symbols.findIndex((s) => s.id === symbol!.id)
   544→      if (idx >= 0) {
   545→        symbols[idx] = { ...symbols[idx], lastCycleTime: Date.now() }
   546→      }
   547→    }
   548→
   549→    // -------------------------------------------------------------------
