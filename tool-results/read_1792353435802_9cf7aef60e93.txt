     1→/**
     2→ * TradeBot Database — Cloudflare-ready (KV) with local JSON file fallback
     3→ *
     4→ * ✓ Works on Cloudflare Workers/Pages using KV namespace (TRADING_DATA)
     5→ * ✓ Works in local dev using JSON file (lib/tradebot/trading_data.json)
     6→ * ✓ ALL methods are async (required for KV)
     7→ *
     8→ * The storage backend is auto-detected at runtime:
     9→ *   - If getRequestContext() from @cloudflare/next-on-pages is available → KV
    10→ *   - Otherwise → JSON file (local dev)
    11→ *
    12→ * KV key layout:
    13→ *   settings:<key>          → individual setting value
    14→ *   automationState:<key>   → individual automation state value
    15→ *   trades                   → JSON array of all trades
    16→ *   positions                → JSON array of all positions
    17→ *   signals                  → JSON array of all signals
    18→ *   logs                     → JSON array of all logs (max 500)
    19→ *   errors                   → JSON array of all errors
    20→ *
    21→ * Each write REPLACES the previous value (no append), so the latest data
    22→ * always overwrites the old data — matching the user's requirement.
    23→ */
    24→
    25→// ---------------------------------------------------------------------------
    26→// KV binding accessor — works with @cloudflare/next-on-pages
    27→// ---------------------------------------------------------------------------
    28→
    29→let _kvCache = undefined // undefined = not yet checked, null = no KV, object = KV
    30→
    31→async function getKV() {
    32→  if (_kvCache !== undefined) return _kvCache
    33→
    34→  try {
    35→    // @cloudflare/next-on-pages provides getRequestContext() to access env bindings
    36→    const { getRequestContext } = await import('@cloudflare/next-on-pages')
    37→    const ctx = getRequestContext()
    38→    _kvCache = (ctx.env && ctx.env.TRADING_DATA) || null
    39→  } catch {
    40→    // Not on Cloudflare — local dev, use JSON file
    41→    _kvCache = null
    42→  }
    43→
    44→  return _kvCache
    45→}
    46→
    47→// ---------------------------------------------------------------------------
    48→// JSON file storage (local dev only)
    49→// ---------------------------------------------------------------------------
    50→
    51→const DB_PATH = '/home/z/my-project/lib/tradebot/trading_data.json'
    52→
    53→let fileDatabase = null
    54→
    55→async function loadFileDatabase() {
    56→  if (fileDatabase !== null) return fileDatabase
    57→  try {
    58→    const fs = await import('fs')
    59→    if (fs.existsSync(DB_PATH)) {
    60→      const data = fs.readFileSync(DB_PATH, 'utf-8')
    61→      fileDatabase = JSON.parse(data)
    62→    } else {
    63→      fileDatabase = {
    64→        settings: {},
    65→        automationState: {},
    66→        trades: [],
    67→        positions: [],
    68→        signals: [],
    69→        logs: [],
    70→        errors: []
    71→      }
    72→      saveFileDatabase()
    73→    }
    74→  } catch (e) {
    75→    console.error('❌ Error loading file database:', e)
    76→    fileDatabase = {
    77→      settings: {},
    78→      automationState: {},
    79→      trades: [],
    80→      positions: [],
    81→      signals: [],
    82→      logs: [],
    83→      errors: []
    84→    }
    85→  }
    86→  return fileDatabase
    87→}
    88→
    89→function saveFileDatabase() {
    90→  if (fileDatabase === null) return
    91→  try {
    92→    // Dynamic import to avoid bundling fs in Cloudflare build
    93→    import('fs').then(fs => {
    94→      fs.writeFileSync(DB_PATH, JSON.stringify(fileDatabase, null, 2))
    95→    })
    96→  } catch (e) {
    97→    console.error('Error saving file database:', e)
    98→  }
    99→}

   101→// ---------------------------------------------------------------------------
   102→// KV helpers
   103→// ---------------------------------------------------------------------------
   104→
   105→async function kvGet(key) {
   106→  const kv = await getKV()
   107→  if (!kv) return null
   108→  const val = await kv.get(key)
   109→  if (val === null || val === undefined) return null
   110→  try {
   111→    return JSON.parse(val)
   112→  } catch {
   113→    return val
   114→  }
   115→}
   116→
   117→async function kvSet(key, value) {
   118→  const kv = await getKV()
   119→  if (!kv) return
   120→  await kv.put(key, JSON.stringify(value))
   121→}

   127→export async function initializeDatabase() {
   128→  const kv = await getKV()
   129→  if (kv) {
   130→    console.log('✅ Database initialized (Cloudflare KV)')
   131→    return true
   132→  }
   133→  await loadFileDatabase()
   134→  console.log('✅ Database loaded successfully (local JSON file)')
   135→  return true
   136→}

   138→// Synchronous version for backward compatibility (no-op in KV/Edge mode)
   139→// In local dev (Node.js), this loads the JSON file synchronously.
   140→// In Edge/Cloudflare, this is a no-op — async initializeDatabase() handles init.
   141→export function initializeDatabaseSync() {
   142→  try {
   143→    // Use globalThis.__tradebotDb to share state with async path
   144→    if (typeof globalThis !== 'undefined' && globalThis.__tradebotFileSyncLoaded) {
   145→      return true
   146→    }
   147→    // Only attempt fs in Node.js (not Edge)
   148→    if (typeof process !== 'undefined' && process.versions && process.versions.node) {
   149→      // eslint-disable-next-line @typescript-eslint/no-var-requires
   150→      const fs = require('fs')
   151→      if (fs.existsSync(DB_PATH)) {
   152→        const data = fs.readFileSync(DB_PATH, 'utf-8')
   153→        fileDatabase = JSON.parse(data)
   154→      } else {
   155→        fileDatabase = {
   156→          settings: {},
   157→          automationState: {},
   158→          trades: [],
   159→          positions: [],
   160→          signals: [],
   161→          logs: [],
   162→          errors: []
   163→        }
   164→        fs.writeFileSync(DB_PATH, JSON.stringify(fileDatabase, null, 2))
   165→      }
   166→      if (typeof globalThis !== 'undefined') {
   167→        globalThis.__tradebotFileSyncLoaded = true
   168→      }
   169→      console.log('✅ Database loaded (sync)')
   170→      return true
   171→    }
   172→    return false
   173→  } catch {
   174→    // Edge Runtime or no fs — async init will handle it
   175→    return false
   176→  }
   177→}

   185→    const kv = await getKV()

   187→      return await kvGet(`settings:${key}`)

   194→    const kv = await getKV()

   197→      return { success: true }
   198→    }
   199→    const db = await loadFileDatabase()
   200→    db.settings[key] = value
   201→    saveFileDatabase()
   202→    return { success: true }
   203→  },

   205→  async getAll() {
   206→    const kv = await getKV()
   207→    if (kv) {
   208→      // KV doesn't support listing with prefix easily in all contexts;
   209→      // return empty object — callers should use individual keys

   211→    }

   214→  },

   595→  // ✓ Batch append — used by /api/run-cycle to write ALL cycle logs in a
   596→  //   SINGLE KV write (read existing logs ONCE, append, write ONCE).
   597→  //   This replaces the previous per-cycle "Strategy 2" write which did a
   598→  //   full read-modify-write on every cycle (one write per cycle).
   599→  //   `newLogs` items should already have: symbol, action, status, message,
   600→  //   error_code, details, created_at. We auto-assign sequential ids.
   601→  async appendBatch(newLogs) {
   602→    if (!Array.isArray(newLogs) || newLogs.length === 0) {
   603→      return { added: 0 }
   604→    }
   605→    const kv = await getKV()
   606→    if (kv) {
   607→      const existing = (await kvGet('logs')) || []
   608→      const maxId = existing.length > 0
   609→        ? Math.max(...existing.map(l => l.id || 0))
   610→        : 0
   611→      const enriched = newLogs.map((l, i) => ({
   612→        id: maxId + i + 1,
   613→        symbol: l.symbol || '',
   614→        action: l.action || 'automation_cycle',
   615→        status: l.status || 'SUCCESS',
   616→        message: l.message || '',
   617→        error_code: l.error_code || null,
   618→        details: l.details || JSON.stringify({ timestamp: l.created_at || new Date().toISOString() }),
   619→        created_at: l.created_at || new Date().toISOString()
   620→      }))
   621→      // Keep only latest 500 logs to stay within KV size limits
   622→      const trimmed = [...existing, ...enriched].slice(-500)
   623→      await kvSet('logs', trimmed)
   624→      return { added: enriched.length }
   625→    }
   626→    // Local dev (JSON file) — fall back to individual pushes
   627→    const db = await loadFileDatabase()
   628→    for (const l of newLogs) {
   629→      l.id = db.logs.length > 0 ? Math.max(...db.logs.map(x => x.id || 0)) + 1 : 1
   630→      l.created_at = l.created_at || new Date().toISOString()
   631→      db.logs.push(l)
   632→    }
   633→    saveFileDatabase()
   634→    return { added: newLogs.length }
   635→  }
   636→}

   650→        message: error.message || '',
   651→        stack: error.stack || '',
   652→        context: error.context || {},
   653→        created_at: new Date().toISOString()
   654→      }
   655→      errors.push(err)
   656→      await kvSet('errors', errors)
   657→      return { id: err.id }
   658→    }
   659→    const db = await loadFileDatabase()
   660→    const err = {
   661→      id: db.errors.length > 0 ? Math.max(...db.errors.map(e => e.id || 0)) + 1 : 1,
   662→      type: error.type || 'UNKNOWN',
   663→      message: error.message || '',
   664→      stack: error.stack || '',
   665→      context: error.context || {},
   666→      created_at: new Date().toISOString()
   667→    }
   668→    db.errors.push(err)
   669→    saveFileDatabase()
   670→    return { id: err.id }
   671→  },

   735→// ---------------------------------------------------------------------------
   736→// Analytics (computed on-the-fly from Trades)
   737→// ---------------------------------------------------------------------------
   738→
   739→export const Analytics = {
   740→  async getTradingStats() {
   741→    const trades = await Trades.getAll(10000)
   742→    const closedTrades = trades.filter(t => t.status === 'CLOSED')
   743→    const wins = closedTrades.filter(t => t.pnl > 0).length
   744→    const losses = closedTrades.filter(t => t.pnl < 0).length
   745→    const totalPnl = closedTrades.reduce((sum, t) => sum + (t.pnl || 0), 0)
   746→
   747→    return {
   748→      total_trades: closedTrades.length,
   749→      winning_trades: wins,
   750→      losing_trades: losses,
   751→      total_pnl: totalPnl,
   752→      avg_pnl: closedTrades.length > 0 ? totalPnl / closedTrades.length : 0,
   753→      max_profit: Math.max(...closedTrades.map(t => t.pnl || 0), 0),
   754→      max_loss: Math.min(...closedTrades.map(t => t.pnl || 0), 0)
   755→    }
   756→  },
   757→
   758→  async getSymbolStats(symbol) {
   759→    const trades = await Trades.getBySymbol(symbol, 10000)
   760→    const closedTrades = trades.filter(t => t.status === 'CLOSED')
   761→    const totalPnl = closedTrades.reduce((sum, t) => sum + (t.pnl || 0), 0)
   762→
   763→    return {
   764→      symbol: symbol,
   765→      total_trades: closedTrades.length,
   766→      winning_trades: closedTrades.filter(t => t.pnl > 0).length,
   767→      total_pnl: totalPnl,
   768→      avg_pnl: closedTrades.length > 0 ? totalPnl / closedTrades.length : 0
   769→    }
   770→  },
   771→
   772→  async getDailyStats(days = 7) {
   773→    const trades = await Trades.getAll(10000)
   774→    const cutoffDate = new Date()
   775→    cutoffDate.setDate(cutoffDate.getDate() - days)
   776→
   777→    const stats = {}
   778→    trades.forEach(trade => {
   779→      if (trade.status === 'CLOSED' && new Date(trade.opened_at) >= cutoffDate) {
   780→        const date = new Date(trade.opened_at).toISOString().split('T')[0]
   781→        if (!stats[date]) {
   782→          stats[date] = { trades: 0, wins: 0, daily_pnl: 0 }
   783→        }
   784→        stats[date].trades++
   785→        if (trade.pnl > 0) stats[date].wins++
   786→        stats[date].daily_pnl += trade.pnl || 0
   787→      }
   788→    })
   789→
   790→    return Object.entries(stats).map(([date, data]) => ({ date, ...data }))
   791→  }
   792→}
